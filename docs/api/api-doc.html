// 浏览器 TypeScript 示例；无第三方依赖。BASE_URL 不包含 /api/v1。
export interface TaskEvent {
  event_id: string;
  event_type: string;
  task_id: string;
  conversation_id: string;
  node_id: string | null;
  payload: Record<string, unknown>;
  created_at: string | null;
}

export class ApiError extends Error {
  constructor(public status: number, public detail: unknown) {
    super(typeof detail === 'string' ? detail : JSON.stringify(detail));
  }
}

export function newId(): string {
  return [...crypto.getRandomValues(new Uint8Array(16))]
    .map((byte) => byte.toString(16).padStart(2, '0')).join('');
}

export const terminalEvents = new Set([
  'agent.run.completed', 'agent.run.failed', 'agent.run.cancelled',
  'task.completed', 'task.failed', 'task.cancelled',
]);

// chunk 不是一条完整事件。按行拆分，保留跨 chunk 的 UTF-8、CRLF 和多行 data。
export async function* decodeEvents(body: ReadableStream<Uint8Array>): AsyncGenerator<TaskEvent> {
  const reader = body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';
  let data: string[] = [];
  let finished = false;
  try {
    while (!finished) {
      const { value, done } = await reader.read();
      finished = done;
      buffer += done ? decoder.decode() : decoder.decode(value, { stream: true });
      while (true) {
        const newline = /\r\n|\r|\n/.exec(buffer);
        if (!newline) break;
        if (!done && newline[0] === '\r' && newline.index === buffer.length - 1) break;
        const line = buffer.slice(0, newline.index);
        buffer = buffer.slice(newline.index + newline[0].length);
        if (line === '') {
          if (data.length) {
            const event: unknown = JSON.parse(data.join('\n'));
            data = [];
            if (typeof event !== 'object' || event === null) throw new Error('Invalid SSE event');
            const candidate = event as Partial<TaskEvent>;
            if (typeof candidate.event_id !== 'string' || !candidate.event_id
              || typeof candidate.task_id !== 'string' || !candidate.task_id
              || typeof candidate.event_type !== 'string' || !candidate.event_type
              || !candidate.payload || typeof candidate.payload !== 'object'
              || Array.isArray(candidate.payload)) throw new Error('Invalid SSE envelope');
            yield candidate as TaskEvent;
          }
        } else if (line.startsWith('data:')) {
          data.push(line.slice(5).replace(/^ /, ''));
        }
        // 注释心跳、id、event 等行无需参与 JSON 拼接；以 envelope 的 event_id/event_type 为准。
      }
    }
    // 未以空行结束的帧不完整，留给调用方通过任务查询/重新订阅恢复。
  } finally {
    if (!finished) await reader.cancel().catch(() => {});
    reader.releaseLock();
  }
}

export function createApi(baseUrl: string) {
  const base = baseUrl.trim().replace(/\/+$/, '');
  let token = '';

  async function check(response: Response): Promise<void> {
    if (response.ok) return;
    const text = await response.text();
    let detail: unknown = text || response.statusText;
    try { detail = JSON.parse(text).detail ?? detail; } catch { /* 代理可能返回文本/HTML。 */ }
    if (response.status === 401) token = '';
    throw new ApiError(response.status, detail);
  }

  async function request<T>(path: string, init: RequestInit = {}): Promise<T> {
    const headers = new Headers(init.headers);
    if (token) headers.set('Authorization', `Bearer ${token}`);
    headers.set('Accept', 'application/json');
    if (init.body && !(init.body instanceof FormData)) headers.set('Content-Type', 'application/json');
    const response = await fetch(`${base}${path}`, { ...init, headers });
    await check(response);
    if (response.status === 204) return undefined as T;
    return response.json() as Promise<T>;
  }

  async function login(username: string) {
    const result = await request<{ access_token: string; user: { username: string } }>(
      '/api/v1/auth/login', { method: 'POST', body: JSON.stringify({ username }) },
    );
    token = result.access_token;
    return result.user;
  }

  async function* events(taskId: string, signal?: AbortSignal): AsyncGenerator<TaskEvent> {
    const response = await fetch(`${base}/api/v1/tasks/${encodeURIComponent(taskId)}/events`, {
      headers: { Authorization: `Bearer ${token}`, Accept: 'text/event-stream' }, signal,
    });
    await check(response);
    if (!response.body || !response.headers.get('Content-Type')?.includes('text/event-stream')) {
      await response.body?.cancel();
      throw new Error('Expected text/event-stream');
    }
    for await (const event of decodeEvents(response.body)) {
      if (event.task_id !== taskId) continue;
      if (event.event_type === 'auth.invalidated') {
        token = '';
        throw new ApiError(401, 'Authentication expired');
      }
      yield event;
    }
  }

  return { request, login, events, setToken(value: string) { token = value; } };
}

// 首次接入：传入测试用户名，onEvent 更新界面。需要中止订阅时传入 AbortController.signal。
// 这是普通对话示例；等待输入时由界面按文档另行提交补参/审批，不能把等待误判为完成。
export async function firstConversation(
  baseUrl: string,
  username: string,
  onEvent: (event: TaskEvent) => void,
  signal?: AbortSignal,
) {
  const api = createApi(baseUrl);
  await api.login(username);
  const conversationId = newId();
  const clientMessageId = newId(); // 同一次消息的网络重试必须复用这个 ID 和原请求体。
  const accepted = await api.request<{ task_id: string; conversation_id: string }>(
    '/api/v1/conversations/chat-messages', {
      method: 'POST', signal,
      body: JSON.stringify({ conversation_id: conversationId, client_message_id: clientMessageId,
        content: '你好，请介绍一下你能帮助我做什么。', routing_mode: 'auto', capability_id: null,
        metadata: { deep_thinking: false } }),
    },
  );
  const seen = new Set<string>(); // 对于后续重新订阅同一 task，请保留此集合。
  let terminal = false;
  let streamError: unknown;
  try {
    for await (const event of api.events(accepted.task_id, signal)) {
      if (seen.has(event.event_id)) continue;
      seen.add(event.event_id);
      onEvent(event);
      if (terminalEvents.has(event.event_type)) { terminal = true; break; }
    }
  } catch (error) {
    if (signal?.aborted || (error instanceof ApiError && [401, 404].includes(error.status))) throw error;
    streamError = error;
  }
  const task = await api.request<{ status: string }>(`/api/v1/tasks/${encodeURIComponent(accepted.task_id)}`);
  if (!terminal && !['completed', 'failed', 'cancelled'].includes(task.status)) {
    throw new Error(`事件流已断开；保存 task_id=${accepted.task_id}，按断线恢复章节重新订阅，不要重新发消息。`, { cause: streamError });
  }
  const [history, artifacts] = await Promise.all([
    api.request<{ messages: Array<{ role: string; content: string; task_id: string | null; stream_status: string | null }> }>(
      `/api/v1/conversations/${encodeURIComponent(conversationId)}/messages`),
    api.request<{ artifacts: unknown[] }>(`/api/v1/tasks/${encodeURIComponent(accepted.task_id)}/artifacts`),
  ]);
  return { conversationId, taskId: accepted.task_id, task, history, artifacts };
}

02 / FOUNDATION

先约定好身份、错误和请求边界

认证与隔离

POST /auth/login 只提交 username；公开能力接口无需认证，其他业务接口都要求 Authorization: Bearer <access_token>。这里的短路径均省略了 /api/v1。登录名为 3–64 位字母、数字、下划线、点或短横线。

  • 身份由 Token 决定，业务请求的 body、metadata、URL 中不要传 username、token、owner 等身份字段。
  • 同一用户名登录或刷新 Token 会替换旧 Token。收到 HTTP 401 或 SSE auth.invalidated 时清理登录态并重新登录。
  • Token 生命周期以服务端当前性校验为准；接口没有独立 refresh_token，也不返回 expires_in，不要自行推算过期时间。
  • 所有业务 ID 按不透明字符串处理；URL 参数用 encodeURIComponent。不要从 task_id、node_id 或 artifact_id 推导业务含义。

浏览器访问与 CORS

优先通过同源 /seedpilot/api/… 代理接入。跨域直连时，服务端需在 MAF_API_CORS_ALLOWED_ORIGINS 配置准确的前端 origin（含协议和端口）；默认不开放跨域，不支持 *，不依赖 Cookie。允许头为 Authorization、Content-Type、Accept。

统一处理三种错误 detail

HTTP 错误可能返回字符串、带 code 的对象或字段校验数组。先读取 HTTP status，再解析 detail；代理错误还可能是纯文本/HTML。

JSON · 三种独立响应形态
[
  {
    "detail": "Authentication required"
  },
  {
    "detail": {
      "code": "message_id_conflict"
    }
  },
  {
    "detail": [

        "loc": [
          "body",

状态前端动作
200 / 201普通成功 / 上传创建成功；读取 JSON 或对应文件流。
202请求被接受。聊天、取消、MCP 测试等仍需跟进实际任务/资源状态。
204成功且无响应体，直接结束，不调用 response.json()。
400 / 422检查业务输入 / DTO 校验。给用户可操作的提示，修正后再提交。
401登录失效。关闭事件流，清理 Token，返回登录。
404资源不存在、已失效或不属于当前用户；刷新列表，不暴露其他用户资源。
409会话忙或 ID/能力状态冲突。按 detail.code 分支处理。
503 / 其他 5xx服务/功能暂不可用。保留用户草稿；先查询已有任务或资源状态，避免盲目重放写请求。

重复点击与网络重试

每次逻辑消息生成一个 client_message_id，提交期间禁用重复发送。同一消息请求超时后保留原 ID 和请求体；服务端可回放既有提交结果。相同 ID 绑定另一条消息/会话会得到 409 message_id_conflict。不要用“生成新 ID 再发一次”恢复已运行的任务。此规则是聊天消息合同，不能扩展为所有写接口都支持幂等重试。

03 / CHAT & TASKS

把聊天、任务和界面状态连接起来

会话不需要单独创建

客户端先生成 conversation_id。第一次发消息（或为新会话上传文件）时由后端建立关联；后续消息沿用同一个 ID。会话列表、消息列表、任务摘要分别负责不同层面的恢复。

三种提交方式

用户意图routing_modecapability_idmetadata
普通聊天，让系统选工具autonull可选 deep_thinking、upload_ids 等。
从 Skill 列表或 /命令选中 Skillhint真实 skill.* ID按需提交附件；无需旧 soft binding 对象。
明确选择 $MCP 服务force_capabilitymcp.dispatch必须含 mcp_server_binding: {server_id}。

auto 携带非空 capability_id、hint/force_capability 缺少 capability_id，或错误的 MCP binding 组合都会被拒绝。Skill hint 表达使用意图，实际行为由主代理结合上下文决定；不可用时处理 409 skill_hint_unavailable 并刷新能力列表。

模型与思考开关

先请求 /config/model-editions。使用 options[].value 作为 model_edition；根据 metadata.deep_thinking 选择 reasoning_efforts.thinking.enabled 或 thinking.disabled 中的 supported/default，把选中的值放到 metadata.main_agent_reasoning_effort。非法组合会返回 validation error，不能假设后端自动降级。未指定选项时采用后端默认配置。

提交结果如何处理

action / 结果界面处理
task_accepted保存 task_id；开始 SSE。status=accepted 不代表有最终回答。
interrupt_resumed继续原 task_id；刷新等待项并确保原任务 SSE 已订阅。
interrupt_clarification_answer展示 assistant_message；保持补充输入状态。
interrupt_mixed_processed / interrupt_schema_switched同时检查 answer_payload.will_resume 和 requires_confirmation；重新读取 open interrupts。
will_resume=false / requires_confirmation=true等待仍未结束。不要把输入框切回普通新任务模式。

刷新恢复与取消

  1. 登录态校验通过后，加载会话列表，选择一个会话。
  2. 同时加载 messages 和 tasks?scope=unfinished。先展示历史消息，再按 task_id 恢复活动任务。
  3. 对活动任务查询摘要、读取当前 interrupts、订阅 SSE；按 message_id / event_id 合并，防止重复气泡。
  4. Task status 是 accepted、planning、running、cancelling、completed、failed、cancelled。等待输入属于 UI 状态，由事件和 open interrupt 驱动；没有独立 waiting Task status。
  5. 用户停止任务时 POST /tasks/cancel。保持“正在取消”直到观察到终态。仅 AbortController.abort() 会关闭本地订阅，不会停止后台任务。

任务完成后用 messages 的最终 assistant 内容和 artifacts 校准界面。失败/取消也要读取已保存的状态，清除陈旧审批框。任务图只是展示快照，当前 edges=[]，不能依赖旧版 DAG 边推导实时进度。

04 / STREAMING

正确处理 SSE、重连和最终结果

使用 fetch 流读取，以便携带 Bearer Token。不要把一个网络 chunk 当作一条事件，也不要把连接结束当作任务完成。

一条真实格式的事件

SSE
id: evt_example
event: agent.run.waiting
data: {"event_id": "evt_example", "conversation_id": "conv_example", "task_id": "task_example", "node_id": null, "event_type": "agent.run.waiting", "payload": {"interrupt_id": "interrupt_example", "reason_kind": "skill_input", "remaining_count": 1}, "created_at": "2026-09-24T10:00:00"}

核心 envelope 字段为 event_id、event_type、task_id、conversation_id、node_id、payload、created_at。使用完整 TypeScript 示例里的 decodeEvents 处理 UTF-8、跨 chunk 换行、多行 data 与注释心跳。浏览器原生 EventSource 不能直接设置本接口需要的 Authorization 头。

最先实现这些事件

事件payload / 前端动作
task.accepted / task.graph_created展示任务已受理；图是当前展示快照。
agent.reasoning_deltadelta、ordinal、sample_id：以 sample_id 分组追加临时思考文字，按 ordinal 去重。它不是最终回答。
agent.reasoning_resetsample_id：清理对应采样的临时思考文字。
agent.run.waitinginterrupt_id、reason_kind、remaining_count：读取 open interrupts，呈现补参/审批。reason_kind 为 skill_input、mcp_approval、mcp_elicitation 或 mcp_remote_task。
agent.run.resumedoutcome、remaining_count：还有等待项时继续等待，不要清掉别的 interrupt。
agent.run.completed / failed / cancelled正式运行终态。公共计数字段为 compaction_count、duration_seconds、outcome、sample_count、tool_call_count；failed 可含 code（例如 model_unavailable）。
task.completed / failed / cancelled也属于任务终态；与 agent.run.* 终态统一处理，避免重复收尾。
node.started / completed / failed更新工具执行概览。单个 node.completed 不等于整个任务完成。
node.waiting_for_input / ready_to_resume / resuming维护对应节点/interrupt 的等待与恢复展示。
task.cancellation_requested / node.cancelled / node.blocked_by_cancellation进入取消过程或更新节点状态，仍以任务终态收尾。
auth.invalidatedreason、connection_id 等：停止重连，清理 Token并重新登录。

一次订阅的生命周期

订阅 → 历史重放 → 实时事件 → 可能等待/恢复 → 任务终态 → 加载历史与产物
  • 每次订阅都会先重放已持久化的 frontend 事件。当前不提供 Last-Event-ID 或 after_event_id 续传参数。保留同一 task 的 event_id 去重集合，接收事件先校验 task_id。
  • agent.reasoning_delta / reset 仅实时传输，不入历史，不会在重连时补齐。清理中断采样的临时显示，最终 answer 从消息历史获得。
  • 不要等连接关闭才更新 UI：收到终态后可关闭订阅并收尾。未知事件可忽略，已认识的结构不合法时触发状态校准，不猜测字段含义。
  • 流意外结束/网络失败：GET /tasks/{task_id}。若已终态，读取 messages/artifacts；若仍活动，在 UI 显示重连提示后延迟重新订阅。401/404 不应无限重试。
  • 同一界面切换会话时关闭旧订阅，并校验会话/订阅代次，避免迟到事件覆盖新会话。
  • 订阅也承载 MCP 在线租约；操作 MCP 等待/审批时保持或恢复原任务订阅。

更多进度事件：skill.progress、capability.missing_fallback、memory.reasoning_delta、interrupt.reasoning_delta。后两者是对应过程的临时展示信息；不要把 audit-only 的内部 Agent 采样/工具日志作为公开 UI 合同。

05 / WAIT & RESUME

补充输入和审批：继续同一个任务

收到等待事件，或恢复时发现 open interrupt，先调用 interrupts 接口。以 interrupt_id 区分等待项，展示 question，再通过同一个 chat-messages 接口回答。

普通 Skill 补参 / 文件选择歧义content 写自然语言回答；附件 ID 放 metadata.upload_ids。

MCP 表单输入metadata.mcp_input_responses 为按 required_fields 组装的对象。
远端 MCP Task 取消输入仅对应 mcp_remote_task_input_required：metadata.mcp_remote_task_cancel=true。

所有回答都带 metadata.interrupt_id；同一次回答重试保留相同 client_message_id。required_fields 中的 slot collection revision、内部 locator 和以下划线开头的私有上下文不应由客户端构造或回传。非法答案返回 400 后保持等待，允许用户修正。

审批/补参的 HTTP 响应可能晚于新的 SSE 事件。开始提交前确保原 task 的 SSE 活跃；用 task_id + interrupt_id 对齐状态。旧响应不能关闭后来出现的新审批，也不能把已完成/取消的任务恢复成等待。

always_allow 只授权该服务的具体 Tool，不代表授权整个 Server。遇到追问/澄清时检查 assistant_message 和 answer_payload：回答被接受也可能继续保持 open interrupt。

07 / OPTIONAL MCP

按需接入 MCP 服务与工具授权

普通聊天先跑通即可；需要让用户配置外部服务时再接这一组接口。所有 Server、Grant 和控制接口都受当前用户 Token 隔离。

设置页面流程

  1. GET /mcp/servers 获取配置列表，显示 enabled、health_status、credential_configured。
  2. POST 添加或 PATCH 编辑。创建/测试返回 202 后查询详情，直到 testing 结束；不能用 HTTP 成功代替健康状态。
  3. 删除服务返回 204 时结束；返回 202 deletion_pending 时展示“正在删除”并刷新确认。
  4. GET /mcp/grants 展示长期 Tool 授权，可删除单条或清空某个 Server 的授权。

凭据如何提交

auth_type请求字段
none省略 credential。
bearercredential: {secret_value: "…"}。
api_key_headerauth_metadata: {header_name: "X-Api-Key"}，credential: {secret_value: "…"}。
static_headerscredential: {static_headers: {"X-Api-Key": "…"}}。
PATCHcredential_action 默认 retain；replace 配合新 credential；clear 需配合合理的 auth_type。

不回显凭据，也不要把“未修改”保存成空密钥。endpoint_url 受公网地址策略约束，私网/特殊地址会被拒绝。公网 HTTP 在当前实现可被接受，但不加密；配置 UI 应在用户选择明文 HTTP 时明确确认。协议通常用 auto；当前接受 2024-11-05、2025-03-26、2025-06-18、2025-11-25、2026-07-28，具体服务仍需连接测试。

文献知识库

管理入口为“用户账户设置 → 文献知识库”。用户可创建库并填写描述，批量上传已有 Markdown；系统提取全文信息后，所有者审核标题、作者、年份、DOI、关键词与检索简述。仅人工确认并完成发布的文献进入索引和 Agent 可读集合。

建库 → POST imports（files 多文件）→ 查询/修正草稿 → confirm → 阅读索引与当前全文

HTTP 202 表示接收导入，未表示入库完成。按每条结果的 status、publication_state、cleanup_pending 和 error_code 展示处理、失败或清理未完成状态。批量确认逐篇返回结果；修订冲突后先刷新。替换正文必须指定 replacement_document_id 并重新审核；成功后文献 ID 不变，旧正文被清除。相同原始字节 SHA-256 在同库复用现有文献或草稿，跨库独立保存。

共享仅接受已有用户名并立即授予 reader。reader 可查看索引、阅读全文、导出与问答；只有 owner 能维护、查看草稿和管理共享。所有读取与下载均校验当前授权；下载字节可用 X-Content-SHA256 校验。

输入区域默认“全部可访问库”；selected 模式至少选择一个库。偏好按会话保存，首次发消息前也可保存；同会话保存按用户操作顺序串行，新问题提交前等待最近选择保存成功，保存失败保留输入并阻止按钮及回车发送。运行中修改只影响下一次提问；重试、恢复保持本轮原候选范围，撤权和删除仍立即生效。部分失效使用剩余指定库并提示；全部失效要求重选，禁止回退为全部库。

Agent 使用三个独立 knowledge 动作，经描述选库、索引选文献、阅读全文后作答。正文和索引工具分页不会静默截断为全文；当前范围外的历史内容不能作为本轮文献依据。输出后的聊天答案独立保留。
